To persist budget changes and monthly income, run this query in your Supabase Dashboard → SQL Editor.

create table if not exists budget_settings (
    id integer primary key default 1 check (id = 1),
    monthly_income numeric(12, 2) not null default 0.00,
    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now()
);
insert into budget_settings (id, monthly_income) values (1, 0.00) on conflict (id) do nothing;
alter table budget_settings enable row level security;

create table if not exists budgets (
    id uuid primary key default uuid_generate_v4(),
    category text not null unique references categories(name) on update cascade on delete cascade,
    avg_spending numeric(12, 2) not null default 0.00,
    proposed_budget numeric(12, 2) not null default 0.00,
    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now()
);
alter table budgets enable row level security;

with category_avg as (
    select category, coalesce(round(avg(total_amount), 2), 0.00) as avg_monthly
    from monthly_category_summary group by category
)
insert into budgets (category, avg_spending, proposed_budget)
select c.name, coalesce(ca.avg_monthly, 0.00), coalesce(ca.avg_monthly, 0.00)
from categories c left join category_avg ca on ca.category = c.name
where c.type = 'variable' and c.is_active = true
on conflict (category) do update set avg_spending = excluded.avg_spending;
